- lib/payments/actions.ts:
  - type PaymentAction = 'capture' | 'retry' | 'release' | 'waive' | 'send_link'
  - allowedPaymentActions(booking: Record<string, unknown>): PaymentAction[]
  - toPaymentFacts(booking): the GET payment object of the contract (pickup_at via pickupInstantMs, completed = status 'completed')
  - async runPaymentAction(booking, action, admin: { id: string }): Promise<{ status: 200; outcome: 'ok' | 'declined' } | { status: 409; error: 'payment_busy' } | { status: 422; error: 'action_not_allowed' }>
    - capture -> settleBooking: captured -> ok; failed / balance_due -> declined; busy / transient -> 409
    - retry: completed -> settleBooking; else claimForHold (not applied -> 409) then placeHold: authorized -> ok; failed -> declined; transient -> 409
    - release -> releaseHold(reason 'operator'); waive -> releaseHold(reason 'waive'): released -> ok; busy -> 409; noop -> 422
    - send_link -> refreshRecoveryLink(booking, payment_amount_minor - captured_amount_minor, 'admin-' + Date.now()) then sendPaymentEmail payment_failed (variant declined, eventKey 'email:payment_failed:<id>:admin:<tag>') -> ok
    - every action first appends admin_action (key 'admin:<action>:<id>:<ms>', actor { type:'admin', id })
</interfaces>
