- table public.identity_admin_log(id uuid PK DEFAULT gen_random_uuid(), action text NOT NULL CHECK (action IN ('split','mark_shared')), actor uuid NOT NULL, identity_id uuid, created_at timestamptz NOT NULL DEFAULT now()) — ids only; RLS deny-all + REVOKE ALL from anon, authenticated.
- public.identity_phone_merges() RETURNS TABLE(identity_id uuid, contact_email text, merged_email text, phone text, bookings integer, merged_at timestamptz): one row per identity_emails row with basis 'phone'; contact_email = the identity's earliest email whose basis is not 'phone'; bookings = booking_identity rows of the identity whose booking's normalised client_email equals merged_email; ORDER BY merged_at DESC.
- public.identity_shared_phones() RETURNS TABLE(phone text, source text, note text, marked_at timestamptz, emails text[]): phone_flags rows with shared = true; emails = sorted distinct identity_emails of identities holding the phone in identity_phones plus rows whose via_phone is the phone; ORDER BY marked_at DESC.
- public.identity_mark_shared(p_phone text, p_note text, p_actor uuid) RETURNS text: 'invalid' unless p_phone matches `^\+[1-9][0-9]{7,14}$`; 'already_shared' when a shared row exists; else INSERT (p_phone, true, 'manual', left(p_note, 80), p_actor), log row, 'marked'.
- public.identity_split(p_email text, p_actor uuid) RETURNS jsonb { outcome, old_identity_id, new_identity_id, keys }: take the global identity lock; e := identity_normalize_email(p_email); no row -> 'not_found'; basis <> 'phone' -> 'not_phone_merge'; old identity erasing -> 'erasing'; create identity N; UPDATE identity_emails SET identity_id = N, basis = 'split' for e (via_phone kept for history); move booking_identity rows of the old identity whose booking's identity_normalize_email(client_email) = e to N; when the old identity's user_id has auth email e, move user_id to N; INSERT phone_flags(via_phone, true, 'split') ON CONFLICT (phone) DO UPDATE SET source = 'split' WHERE phone_flags.source <> 'manual'; log row; for N and old: outbox_enqueue('sync.espo'|'sync.chatwoot', 'person', id, '<type>:<id>:s' || floor(extract(epoch FROM clock_timestamp()))::bigint, now()); return outcome 'split' and the four keys.
